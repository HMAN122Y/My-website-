function mcq(host,st,q,opts,onUpdate){
  host.innerHTML='<p class="q"></p><div class="opts"></div><div class="fb" aria-live="polite"></div><div class="retry"></div>';
  host.querySelector(".q").textContent=q;
  var box=host.querySelector(".opts"),fb=host.querySelector(".fb"),rt=host.querySelector(".retry");
  function paint(){
    box.innerHTML="";fb.className="fb";fb.textContent="";rt.innerHTML="";
    opts.forEach(function(o,i){
      var b=document.createElement("button");b.type="button";b.className="opt";
      if(o.i){var ic=document.createElement("span");ic.className="oi";ic.textContent=o.i;ic.setAttribute("aria-hidden","true");b.appendChild(ic);}
      var tx=document.createElement("span");tx.textContent=o.t;b.appendChild(tx);
      if(st.done){b.disabled=true;if(o.ok)b.classList.add("ok");}
      else if(st.sel!=null){b.disabled=true;if(st.sel===i)b.classList.add("no");}
      b.onclick=function(){choose(i);};box.appendChild(b);
    });
    if(st.done){var c=opts.filter(function(o){return o.ok;})[0];fb.className="fb good";fb.textContent="🎉 "+c.why;}
    else if(st.sel!=null){
      fb.className="fb bad";fb.textContent="💡 "+opts[st.sel].why;
      var r=document.createElement("button");r.type="button";r.className="btn small";r.textContent="أعد المحاولة";
      r.onclick=function(){st.sel=null;paint();};rt.appendChild(r);
    }
  }
  function choose(i){
    if(st.done)return;st.tries=(st.tries||0)+1;
    if(opts[i].ok){st.done=true;st.first=(st.tries===1);st.sel=null;}else{st.sel=i;}
    paint();if(onUpdate)onUpdate(st);
  }
  paint();
}

  "void main(){vec2 q=(gl_FragCoord.xy-res*.5)/res.y;float f=.5/tan(fov*.5);vec3 d=normalize(vec3(q.x,q.y,f));"+
  "float cp=cos(pitch),sp=sin(pitch);d=vec3(d.x,d.y*cp+d.z*sp,-d.y*sp+d.z*cp);"+
  "float cy=cos(yaw),sy=sin(yaw);d=vec3(d.x*cy+d.z*sy,d.y,-d.x*sy+d.z*cy);"+
  "float lon=atan(d.x,d.z);float lat=asin(clamp(d.y,-1.,1.));"+
